CREATE TABLE UNIVERSITY (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    location VARCHAR(100),
    year INT
);

CREATE TABLE DEPARTMENT (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    hod VARCHAR(100),
    university_id INT,
    FOREIGN KEY (university_id)
        REFERENCES UNIVERSITY(id)
);

CREATE TABLE PROGRAM(
    id INT PRIMARY KEY,
    name VARCHAR(100),
    duration VARCHAR(100),
    university_id INT,
    FOREIGN KEY (university_id)
        REFERENCES UNIVERSITY(id)
);

CREATE TABLE SYLLABUS(
    id INT PRIMARY KEY,
    topic VARCHAR(100),
    university_id INT,
    FOREIGN KEY (university_id )
        REFERENCES UNIVERSITY(id)
);

CREATE TABLE COURSE(
    id INT PRIMARY KEY,
    topic VARCHAR(100),
    program_id INT,
    syllabus_id INT,
    FOREIGN KEY (program_id )
        REFERENCES PROGRAM(id),
    FOREIGN KEY(syllabus_id)
        REFERENCES SYLLABUS(id)
);

CREATE TABLE FACULTY(
    id INT PRIMARY KEY,
    name VARCHAR(100),
    specialisation VARCHAR(100),
    university_id INT,
    FOREIGN KEY (university_id)
        REFERENCES UNIVERSITY(id)
);

CREATE INDEX idx_department_university
ON DEPARTMENT(university_id);

CREATE INDEX idx_program_university
ON PROGRAM(university_id);

CREATE INDEX idx_syllabus_university
ON SYLLABUS(university_id);

CREATE INDEX idx_course_program
ON COURSE(program_id);

CREATE INDEX idx_course_syllabus
ON COURSE(syllabus_id);

CREATE INDEX idx_faculty_university
ON FACULTY(university_id);

FROM UNIVERSITY
WHERE location = 'Delhi';

FROM DEPARTMENT D
JOIN UNIVERSITY U
    ON D.university_id = U.id
WHERE U.name = 'AMU';

FROM UNIVERSITY

SELECT U.name
FROM UNIVERSITY U
JOIN PROGRAM P
    ON U.id = P.university_id
WHERE P.name = 'MCA';


SELECT C.*
FROM COURSE C
JOIN PROGRAM P
    ON C.program_id = P.id
JOIN UNIVERSITY U
    ON P.university_id = U.id
WHERE P.name = 'MCA'
  AND U.name = 'AMU';



SELECT F.name AS faculty_name,
       F.specialisation,
       U.name AS university_name
FROM FACULTY F
JOIN UNIVERSITY U
    ON F.university_id = U.id
WHERE F.specialisation = 'CS';
