CREATE TABLE PERSON(driver_id VARCHAR(4) PRIMARY KEY, name VARCHAR(20), address VARCHAR(100));
CREATE TABLE CAR(regno VARCHAR(10) PRIMARY KEY, model VARCHAR(20), year INT);
CREATE TABLE OWNS(driver_id VARCHAR(4) ,regno_number VARCHAR(10), PRIMARY KEY(driver_id, regno_number) FOREIGN KEY driver_id REFERENCES PERSON(driver_id), FOREIGN KEY regno REFERENCES CAR(regno));
CREATE TABLE OWNS(
        driver_id VARCHAR(4) ,
        regno_number VARCHAR(10),
        PRIMARY KEY (driver_id, regno_number),
        FOREIGN KEY (driver_id) REFERENCES PERSON(driver_id),
        FOREIGN KEY (regno_number) REFERENCES CAR(regno)
);
CREATE TABLE PARTICIPATED(
        driver_id VARCHAR(4) ,
        regno_number VARCHAR(10),
        report_number VARCHAR(10),
        damage_amount INT,
        PRIMARY KEY (driver_id, regno_number),
        FOREIGN KEY (driver_id) REFERENCES PERSON(driver_id),
        FOREIGN KEY (regno_number) REFERENCES CAR(regno),
        FOREIGN KEY (report_number) REFERENCES ACCIDENT(report_number)
);

FROM OWNS
JOIN PARTICIPATED
    ON PARTICIPATED.driver_id = OWNS.driver_id
    AND PARTICIPATED.regno_number = OWNS.regno_number
JOIN ACCIDENT
    ON ACCIDENT.report_number = PARTICIPATED.report_number
WHERE EXTRACT(YEAR FROM ACCIDENT.accd_date) = 2026;
